$$
\frac {\tan \alpha \tan 2 \alpha}{\tan 2 \alpha - \tan \alpha} + \sqrt {3} (\sin^ {2} \alpha - \cos^ {2} \alpha) = 2 \sin \left(2 \alpha - \frac {\pi}{3}\right). \tag {2}
$$
