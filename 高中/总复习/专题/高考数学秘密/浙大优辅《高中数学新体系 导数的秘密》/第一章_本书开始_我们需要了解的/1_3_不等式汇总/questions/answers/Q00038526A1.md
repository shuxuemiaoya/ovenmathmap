> > a ^ {a} b ^ {b} c ^ {c} + a ^ {b} b ^ {c} c ^ {a} + a ^ {c} b ^ {a} c ^ {b} \leqslant a ^ {2} + b ^ {2} + c ^ {2} + a b + b c + c a + a c + b a + b c = (a + b + c) ^ {2} = 1.
> > $$
