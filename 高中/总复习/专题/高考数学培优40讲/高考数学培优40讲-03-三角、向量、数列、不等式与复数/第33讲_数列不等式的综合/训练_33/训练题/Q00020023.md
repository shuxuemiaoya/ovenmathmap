$$
\frac {1}{2} + \frac {1}{3} + \dots + \frac {1}{2 0 2 3} <   \ln 2 0 2 3 <   1 + \frac {1}{2} + \dots + \frac {1}{2 0 2 2}. \tag {2}
$$
