> [!faq]- 全练一本通解析
> **【解析】**
> $$
\overrightarrow {A B} = \vec {a}, \overrightarrow {A D} = \vec {b}, \overrightarrow {A A ^ {\prime}} = \vec {c}
$$
$$
A B = A A ^ {\prime} = A D
$$
$$
\text {以} \left| \vec {a} \right| = \left| \vec {b} \right| = \left| \vec {c} \right|, \text {令} \left| \vec {a} \right| = \left| \vec {b} \right| = \left| \vec {c} \right| = m (m > 0)
$$
$$
\overrightarrow {B M} = \frac {1}{4} \overrightarrow {B C}
$$
$$
\overrightarrow {D M} = \overrightarrow {D A} + \overrightarrow {A B} + \overrightarrow {B M} = - \overrightarrow {A D} + \overrightarrow {A B} + \frac {1}{4} \overrightarrow {B C} = - \vec {b} + \vec {a} + \frac {1}{4} \vec {b} = \vec {a} - \frac {3}{4} \vec {b}
$$
$$
\overrightarrow {D ^ {\prime} N} = \lambda \overrightarrow {D ^ {\prime} C ^ {\prime}}
$$
$$
\overrightarrow {A N} = \overrightarrow {A A ^ {\prime}} + \overrightarrow {A ^ {\prime} D ^ {\prime}} + \overrightarrow {D ^ {\prime} N} = \overrightarrow {A A ^ {\prime}} + \overrightarrow {A D} + \lambda \overrightarrow {D ^ {\prime} C ^ {\prime}} = \vec {c} + \vec {b} + \lambda \vec {a}
$$
$$
D M \perp A N
$$
$$
\overrightarrow {D M} \cdot \overrightarrow {A N} = 0
$$
$$
\left(\vec {a} - \frac {3}{4} \vec {b}\right) \cdot \left(\vec {c} + \vec {b} + \lambda \vec {a}\right) = 0
$$
$$
\vec {a} \cdot \vec {c} + \vec {a} \cdot \vec {b} + \lambda \vec {a} ^ {2} - \frac {3}{4} \vec {b} \cdot \vec {c} - \frac {3}{4} \vec {b} ^ {2} - \frac {3}{4} \lambda \vec {a} \cdot \vec {b} = 0
$$
$$
\angle B A D = \angle B A A ^ {\prime} = \angle D A A ^ {\prime} = 6 0 ^ {\circ}, \left| \vec {a} \right| = \left| \vec {b} \right| = \left| \vec {c} \right| = m (m > 0)
$$
$$
\frac {1}{2} m ^ {2} + \frac {1}{2} m ^ {2} + \lambda m ^ {2} - \frac {3}{8} m ^ {2} - \frac {3}{4} m ^ {2} - \frac {3}{8} \lambda m ^ {2} = 0
$$
$$
\frac {1}{2} + \frac {1}{2} + \lambda - \frac {3}{8} - \frac {3}{4} - \frac {3}{8} \lambda = 0
$$
$$
\begin{array}{c} \frac {2}{3} \end{array}
$$
$$
\lambda = \frac {1}{5}
$$
