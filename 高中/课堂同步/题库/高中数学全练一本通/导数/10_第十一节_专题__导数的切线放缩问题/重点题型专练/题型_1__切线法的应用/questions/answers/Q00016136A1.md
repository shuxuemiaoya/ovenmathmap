> > $$

> > $$

> > $$

> > $$
> > \forall x \geq 0, a x > - e ^ {x}

> > $$
> > $$
> > f (0) = 1 > 0 \text { 成立 }
> > $$
> > $$

> > $$
> > $$

> > $$
> > $$
> > g (x) = - \frac {e ^ {x}}{x}
> > $$
> > $$
> > g ^ {\prime} (x) = - \frac {e ^ {x} (x - 1)}{x ^ {2}}, x > 0

> > \therefore g (x) \text {在} (0, 1)
