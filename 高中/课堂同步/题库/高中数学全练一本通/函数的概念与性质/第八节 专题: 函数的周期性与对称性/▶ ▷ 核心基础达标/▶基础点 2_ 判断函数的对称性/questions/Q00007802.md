【2】根据结论判断下列抽象函数的对称性：

$$
① f (3 + x) = f (3 - x);
$$

$$
② f (2 + x) = f (4 - x);
$$

$$
③ f (3 + x) - f (- x - 7) = 0;
$$

$$
④ f (1 + x) - f (3 - x) = 0;
$$

$$
⑤ f (4 + 3 x) = f (- 6 - 3 x);
$$

$$
⑥ f (2 x) = f (- 6 - 2 x);
$$

$$
⑦ f (4 + x) + f (4 - x) = 0;
$$

$$
⑧ f (x) + f (- 2 - x) = 0;
$$

$$
⑨ f (1 + x) = - f (5 - x);
$$

$$
⑩ f (- x) + f (x - 6) = 0;
$$

$$
⑪ f (2 x) + f (4 - 2 x) = 0;
$$

$$
f (1 - 3 x) = - f (7 + 3 x) \tag {⑫}
$$

$$
f (3 + x) + f (3 - x) = 4 \tag {13}
$$

$$
f (x) + f (6 - x) = 2 \tag {14}
$$

$$
f (1 + x) = 4 - f (3 - x); \tag {⑮}
$$

$$
f (2 x) = 4 - f (- 6 - 2 x). \tag {16}
$$
