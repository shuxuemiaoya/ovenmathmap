> > = P (X = 0) = \frac {C _ {8} ^ {3}}{C _ {1 0} ^ {3}} = \frac {5 6}{1 2 0} = \frac {7}{1 5}, P (X = 1) = \frac {C _ {2} ^ {1} \times C _ {8} ^ {2}}{C _ {1 0} ^ {3}} = \frac {5 6}{1 2 0} = \frac {7}{1 5}, P (X = 2) = \frac {C _ {2} ^ {2} \times C _ {8} ^ {1}}{C _ {1 0} ^ {3}} = \frac {8}{1 2 0} = \frac {1}{1 5}
