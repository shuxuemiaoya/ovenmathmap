$$
\mathrm{A} \omega = 4 \pi , \varphi = \frac {\pi}{4}
$$

$$
\text {B} \omega = 4 \pi , \varphi = \frac {\pi}{3}
$$

$$
\text {C} \omega = 8 \pi , \varphi = \frac {\pi}{4}
$$

$$
D \omega = 8 \pi , \varphi = \frac {\pi}{3}
$$
