\triangle A B C

a, b, c, 2 c \sin B = (2 a + c) \tan C

$$
2 \sin C \sin B = (2 \sin A + \sin C) \frac {\sin C}{\cos C}
$$
$$
C \in (0, \pi)
$$
$$
\sin C \neq 0
$$
$$
\therefore 2 \sin B \cos C = (2 \sin A + \sin C) = 2 \sin (B + C) + \sin C
$$
$$
= 2 \sin B \cos C + 2 \cos B \sin C + \sin C
$$
$$
\therefore 2 \cos B \sin C + \sin C = 0
$$
$$
\because \sin C \neq 0, \therefore \cos B = - \frac {1}{2}, B \in (0, \pi)
