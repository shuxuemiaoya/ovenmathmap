> >
> > $$
> > \begin{array}{r l} 26 - 8 y + 3 y & = 16, \\ - 5 y & = - 10, \end{array}

> > $$
> > y = 2 _ {\circ}
> > $$

> >
> > $$
> > x = 5 _ {\circ}
> > $$
