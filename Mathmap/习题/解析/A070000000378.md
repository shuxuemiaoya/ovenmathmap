> >
> > $$
> > \therefore \angle B A E = \angle D C F _ {\circ}
> > $$

> >
> > $$
> > \therefore \quad \triangle A B E \cong \triangle C D F _ {\circ}
> > $$
> >
> > $$
> > \therefore B E = D F _ {\circ}
> > $$
