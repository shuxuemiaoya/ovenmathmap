$$

(1) 8 + (- 3) ^ {2} \times (- 2);

$$

(2) 100 \div (- 2) ^ {2} - (- 2) \div \left(- \frac {2}{3}\right);

$$

(3) (- 4) \div \left(- \frac {3}{4}\right) \times (- 3);
$$

$$
(- \frac {1}{3}) \div (- \frac {1}{3}) ^ {2} - 4 \times (- \frac {1}{2}) ^ {3} 。 \tag {4}
